# Probe source (main API):
//! Probe of main-branch behaviour for the A2 red/green record (not committed).
use super::tests::{make_signed_changeset, temp_store_dir};
use super::*;
use ed25519_dalek::SigningKey;

fn blob_files(dir: &Path) -> usize {
    fn walk(p: &Path) -> usize {
        let Ok(rd) = fs::read_dir(p) else { return 0 };
        rd.map(|e| {
            let e = e.unwrap();
            if e.file_type().unwrap().is_dir() { walk(&e.path()) } else { 1 }
        })
        .sum()
    }
    walk(&dir.join("blobs"))
}

#[test]
fn a2_checkpoint_keeps_blobs_in_segments_across_reopen() {
    let dir = temp_store_dir("a2-main-probe");
    let key = SigningKey::from_bytes(&[90; 32]);
    let mut s = SynapseStore::open(&dir, FsyncPolicy::GROUP_COMMIT_DEFAULT,
        RecoveryPolicy::StrictFailClosed, StoreLimits::default()).unwrap();
    for i in 0..20 {
        let label = format!("ck-{i}");
        let (env, blobs) = make_signed_changeset(&key, &label, &format!("src/{label}.rs"),
            &vec![b'a' + (i as u8 % 26); 6000], vec![], 1000);
        s.ingest_changeset(env, &blobs).unwrap();
    }
    let v2_before = fs::read(dir.join(wal2::WAL_V2_FILE)).unwrap();
    s.checkpoint_and_compact(1000).unwrap();
    let v2_after = fs::read(dir.join(wal2::WAL_V2_FILE)).unwrap();
    let kept = v2_after.len() >= v2_before.len() && v2_after[..v2_before.len()] == v2_before[..];
    assert_eq!(blob_files(&dir), 0, "no file per blob (main wrote {} blob files; log kept whole: {kept})", blob_files(&dir));
}
