+    /// CRC32C and hash chain, so a resealed forgery) without the owner's key, must change
+    /// nothing on replay.
+    #[test]
+    fn b89_forged_owner_move_resealed_in_the_wal_changes_nothing_on_replay() {
+        use crate::{FsyncPolicy, RecoveryPolicy, StoreLimits, SynapseStore};
+        let dir = std::env::temp_dir().join(format!("synapse-b89-red-{}", std::process::id()));
+        let _ = std::fs::remove_dir_all(&dir);
+        let open = || {
+            SynapseStore::open(
+                &dir,
+                FsyncPolicy::EveryCommit,
+                RecoveryPolicy::TruncateTornTail,
+                StoreLimits::default(),
+            )
+            .unwrap()
+        };
+        let mut s = open();
+        let mut rng = SplitMix64(0xB89);
+        let (envelope, owner) = match admission(&mut rng, 1) {
+            WalRecord::RevisionAdmitted { envelope, chain, .. } => (envelope, chain),
+            _ => unreachable!(),
+        };
+        let body = 1u32.to_le_bytes();
+        let blobs = vec![(BlobDigest::compute(&body), body.to_vec())];
+        let id = envelope.changeset_id();
+        s.admit_revision(
+            envelope,
+            &blobs,
+            crate::AdmissionAuthority {

+                change_scope: Vec::new(),
+            },
